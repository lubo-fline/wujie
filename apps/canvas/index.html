Canvas 实验室

《Canvas 背景特效》三章的完整落地：随机粒子、鼠标互动、以及把它们封装成一个带高分屏适配和降帧的类。三个面板共用同一批粒子，区别只在于多做了什么。

01 使粒子动起来 · 一百行的起点

每个粒子就是一个对象：位置、速度、半径。每帧清屏 → 更新位置 → 画圆 → 撞墙反向。所有 Canvas 动画都是这个骨架，后面两章只是在这骨架上加东西。密度滑杆实时增删粒子。

const p = { x: rand(w), y: rand(h), vx: rand(±0.6), vy: rand(±0.6), r: 1 + rand(2) }; function frame() { ctx.clearRect(0, 0, w, h); // ① 清屏 for (const p of parts) { p.x += p.vx; p.y += p.vy; // ② 更新 if (p.x < 0 || p.x > w) p.vx *= -1; // ③ 撞墙反向 ctx.arc(p.x, p.y, p.r, 0, 7); // ④ 画圆 } requestAnimationFrame(frame); }

02 鼠标互动 · 一条距离公式定亲疏

在画面里移动鼠标：靠近的粒子被推开、连线只在阈值内出现、离鼠标越近线越亮。没有新的机制，只是每帧多问每个粒子一句「你离鼠标多远」，然后按距离改速度和透明度。

const dx = mouse.x - p.x, dy = mouse.y - p.y; const dist = Math.hypot(dx, dy); // 排斥：距离越近推力越大，超出范围不受影响 if (dist < 120 && dist > 0.01) { const f = (1 - dist / 120) * 0.6; p.vx -= (dx / dist) * f; p.vy -= (dy / dist) * f; } // 连线：两两判断距离，alpha 随距离衰减 if (dist2 < LINK) { ctx.strokeStyle = `rgba(140,165,255,${1 - dist2 / LINK})`; ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke(); }

03 更优雅的 Canvas · 类封装 + 高分屏 + 降帧

把前两个面板封装成一个 ParticleField 类：构造时传容器，自动适配 devicePixelRatio（不模糊不虚）、支持降帧（省电省 GPU）、页面切走自动暂停、回来自动恢复。右上角的标签实时显示 DPR 和实际帧率——前两个面板在同一块高分屏上是 2 倍物理像素渲染，但没处理 DPR 的话看起来一样，放到 Retina 屏上就会发虚。

class ParticleField { constructor(container, opts) { this.dpr = Math.min(devicePixelRatio || 1, 2); // 关键：canvas 尺寸 = CSS 尺寸 × dpr，再用 ctx.scale 缩回 canvas.width = cssW * this.dpr; canvas.height = cssH * this.dpr; this.ctx.scale(this.dpr, this.dpr); } start() { this.last = 0; const loop = (t) => { // 降帧：目标 30fps 就每两帧画一次 if (t - this.last >= 1000 / this.fps) { this.step(); this.draw(); this.last = t; } this.raf = requestAnimationFrame(loop); }; } } // 页面隐藏自动暂停，不白烧 GPU document.addEventListener('visibilitychange', () => api.paused && api.pause());
